/** * Regional Events Dashboard -> Exhibit Day sync * ---------------------------------------------------------------------------- * This Cloud Function holds the Exhibit Day API key securely on the server * side (via Secret Manager) and proxies sync requests from the dashboard, * which never sees or handles the key itself. * * VERIFY BEFORE FIRST USE: the exact endpoint paths below (EXHIBITDAY_BASE_URL * and the paths in getExistingExhibitDayEvent/createExhibitDayEvent/ * updateExhibitDayEvent) are built from Exhibit Day's documented parameters * using standard REST conventions, but their Swagger UI (https://api.exhibitday.com/swagger/ui/index) * or Postman collection (https://www.postman.com/exhibitday/exhibitday-api/overview) * is the authoritative source -- open one of those, find the exact URL shown * for "Retrieve Events", "Add an Event", and "Update an Event", and adjust the * three spots marked "VERIFY THIS PATH" below if they differ. */ const { onCall, HttpsError } = require('firebase-functions/v2/https'); const { defineSecret } = require('firebase-functions/params'); const admin = require('firebase-admin'); admin.initializeApp(); // Only this account can assign which brand a user's account belongs to -- matches the same // admin allowlist pattern used for the dashboard's own Settings panel. const BRAND_ADMIN_EMAILS = ['paula.shoemaker@greatdayimprovements.com']; // The only valid brand values -- keeps a typo from silently creating a new, unintended brand // that would just look like an empty dashboard to whoever got assigned to it by mistake. const VALID_BRANDS = ['patio-enclosures', 'universal-windows-bath-authority', 'champion-windows', 'leafguard']; const exhibitDayApiKey = defineSecret('EXHIBITDAY_API_KEY'); const EXHIBITDAY_BASE_URL = 'https://api.exhibitday.com/v1'; async function findExistingExhibitDayEvent(dashboardEventId, apiKey) { const url = new URL(`${EXHIBITDAY_BASE_URL}/events`); // VERIFY THIS PATH ("Retrieve Events") url.searchParams.set('api_key', apiKey); url.searchParams.set('filter_by_integration_metadata_field_1', dashboardEventId); const res = await fetch(url.toString()); if (!res.ok) { throw new Error(`Exhibit Day lookup failed (${res.status}): ${await res.text()}`); } const data = await res.json(); const list = Array.isArray(data) ? data : (data.events || data.results || []); return list.length > 0 ? list[0] : null; } async function createExhibitDayEvent(payload, apiKey) { const url = `${EXHIBITDAY_BASE_URL}/events`; // VERIFY THIS PATH ("Add an Event") const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ api_key: apiKey, ...payload }) }); if (!res.ok) { throw new Error(`Exhibit Day create failed (${res.status}): ${await res.text()}`); } return res.json(); } async function updateExhibitDayEvent(exhibitDayEventId, payload, apiKey) { const url = `${EXHIBITDAY_BASE_URL}/events`; // VERIFY THIS PATH ("Update an Event") const res = await fetch(url, { method: 'PATCH', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ api_key: apiKey, id: exhibitDayEventId, ...payload }) }); if (!res.ok) { throw new Error(`Exhibit Day update failed (${res.status}): ${await res.text()}`); } return res.json(); } exports.syncEventToExhibitDay = onCall({ secrets: [exhibitDayApiKey] }, async (request) => { if (!request.auth) { throw new HttpsError('unauthenticated', 'You must be signed in to sync events.'); } const ev = request.data; if (!ev || !ev.dashboardEventId || !ev.name || !ev.start_date || !ev.end_date) { throw new HttpsError('invalid-argument', 'Missing required event fields (dashboardEventId, name, start_date, end_date).'); } const apiKey = exhibitDayApiKey.value(); try { const existing = await findExistingExhibitDayEvent(ev.dashboardEventId, apiKey); const richFieldsPayload = { venue_name: ev.venue_name, website_url: ev.website_url, event_notes: ev.event_notes, budget_notes: ev.budget_notes, budget_booth_reservation: ev.budget_booth_reservation, budget_travel: ev.budget_travel, budget_misc_expenses: ev.budget_misc_expenses, budget_attendee_registrations: ev.budget_attendee_registrations, integration_metadata_field_1: ev.dashboardEventId, integration_metadata_field_2: ev.dashboardStatus || '', ...(ev.custom_fields || {}) }; let result; if (existing) { result = await updateExhibitDayEvent(existing.id, { name: ev.name, start_date: ev.start_date, end_date: ev.end_date, ...richFieldsPayload }, apiKey); result._syncAction = 'updated'; result._exhibitDayEventId = existing.id; } else { const created = await createExhibitDayEvent({ name: ev.name, start_date: ev.start_date, end_date: ev.end_date }, apiKey); result = await updateExhibitDayEvent(created.id, richFieldsPayload, apiKey); result._syncAction = 'created'; result._exhibitDayEventId = created.id; } return result; } catch (err) { console.error('Exhibit Day sync failed for dashboard event', ev.dashboardEventId, err); throw new HttpsError('internal', err.message || 'Sync to Exhibit Day failed.'); } }); exports.listExhibitDayCustomFields = onCall({ secrets: [exhibitDayApiKey] }, async (request) => { if (!request.auth) { throw new HttpsError('unauthenticated', 'You must be signed in to do this.'); } const apiKey = exhibitDayApiKey.value(); const url = new URL(`${EXHIBITDAY_BASE_URL}/references/event_custom_fields`); // VERIFY THIS PATH url.searchParams.set('api_key', apiKey); const res = await fetch(url.toString()); if (!res.ok) { throw new HttpsError('internal', `Custom field lookup failed (${res.status}): ${await res.text()}`); } return res.json(); }); exports.assignUserBrand = onCall({}, async (request) => { if (!request.auth || !request.auth.token.email) { throw new HttpsError('unauthenticated', 'You must be signed in to do this.'); } if (!BRAND_ADMIN_EMAILS.includes(request.auth.token.email.toLowerCase())) { throw new HttpsError('permission-denied', 'Only an administrator can assign brand access.'); } const { email, brand } = request.data || {}; if (!email || !brand) { throw new HttpsError('invalid-argument', 'Both an email and a brand are required.'); } if (!VALID_BRANDS.includes(brand)) { throw new HttpsError('invalid-argument', `"${brand}" isn't a recognized brand. Valid options: ${VALID_BRANDS.join(', ')}`); } try { const user = await admin.auth().getUserByEmail(email); await admin.auth().setCustomUserClaims(user.uid, { brand }); return { success: true, uid: user.uid, email: user.email, brand }; } catch (err) { console.error('Failed to assign brand', email, brand, err); if (err.code === 'auth/user-not-found') { throw new HttpsError('not-found', `No account exists for ${email}. Create the account in Firebase Authentication first, then assign its brand.`); } throw new HttpsError('internal', err.message || 'Could not assign brand.'); } }); exports.getUserBrand = onCall({}, async (request) => { if (!request.auth || !request.auth.token.email) { throw new HttpsError('unauthenticated', 'You must be signed in to do this.'); } if (!BRAND_ADMIN_EMAILS.includes(request.auth.token.email.toLowerCase())) { throw new HttpsError('permission-denied', 'Only an administrator can look this up.'); } const { email } = request.data || {}; if (!email) { throw new HttpsError('invalid-argument', 'An email is required.'); } try { const user = await admin.auth().getUserByEmail(email); return { email: user.email, brand: (user.customClaims && user.customClaims.brand) || null }; } catch (err) { if (err.code === 'auth/user-not-found') { throw new HttpsError('not-found', `No account exists for ${email}.`); } throw new HttpsError('internal', err.message || 'Lookup failed.'); } });
